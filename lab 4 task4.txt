% Prolog Family Task 2

% Finds all female names
% Query asks Prolog to return every person that satisfies female(X).

?- female(X).

X = sara ;
X = nora.


% Finds all males
% Query asks Prolog to return every person that satisfies male(X).

?- male(X).

X = ali ;
X = ahmad ;
X = muhammad ;
X = faisal ;
X = hassan ;
X = khan.


% Finds the grandparent of Faisal
% A grandparent is a person who is the parent of Faisal's parent. grandparent(X,Y) :- parent(X,Z) , parent(Z,Y).

?- grandparent(X,faisal).

X = ali.

% Finds daughters of Ahmad
% A daughter is a female person whose parent is Ahmad daughter(X,Y) :- female(X), parent(Y,X).

?- daughter(X,ahmad).

X = sara ;
X = nora.


% Finds common parent between two people
% Two people have a common parent if they share the same parent. common_parent(X,Y,P) :- parent(P,X), parent(P,Y).

?- common_parent(ahmad,muhammad,P).

P = ali.


% Checks if Khan is the brother of Nora
% A brother is a male person who shares the same parent with another person. brother(X,Y) :- male(X), parent(P,X), parent(P,Y).

?- brother(khan,nora).

true.
